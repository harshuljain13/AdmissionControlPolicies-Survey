% arXiv preprint wrapper. Standalone: not TMLR, not acmart.
%
% This used to build on tmlr.sty, which was left over from when TMLR was
% the target venue. It rendered correctly but put the preprint in a
% specific journal's house style, with five stacked author blocks each
% repeating the same affiliation. A preprint should not look like a
% submission to a venue it is not being submitted to.
%
% Author block follows the usual preprint convention: names on one line,
% shared affiliation stated once rather than five times, emails
% together. Typography matches the CSUR build so the two read alike.

% body.tex uses \rowcolors, which needs xcolor's table option in place
% before xcolor loads. Must precede \documentclass.
\PassOptionsToPackage{table}{xcolor}

\documentclass[11pt]{article}

\usepackage[letterpaper,margin=1in]{geometry}

% Libertine as OpenType, same as the CSUR build. Loaded by filename:
% fontspec cannot resolve the family name in this bundle, and
% \usepackage{libertine} falls back to Computer Modern without erroring.
\usepackage{fontspec}
\setmainfont{LinLibertine_R.otf}[
  BoldFont=LinLibertine_RB.otf,
  ItalicFont=LinLibertine_RI.otf,
  BoldItalicFont=LinLibertine_RBI.otf]
\setsansfont{LinBiolinum_R.otf}[BoldFont=LinBiolinum_RB.otf]
\setmonofont{lmmono10-regular.otf}

\input{../preamble}

\usepackage[numbers,sort&compress]{natbib}
\usepackage{fancyhdr}

% arXiv and Google Scholar read these fields. The PDF previously carried
% neither a title nor an author.
\usepackage{hyperref}
