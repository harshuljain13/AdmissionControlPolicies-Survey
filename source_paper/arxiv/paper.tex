% arXiv preprint wrapper.
%
% Built on the ICLR conference style, which is this group's house format
% for preprints (see arXiv:2608.28363): "Preprint" running head, authors
% in a grid with affiliation and email under each name, Times text with
% Computer Modern math, author-year citations.
%
% Two deliberate departures from that style. Headings and the title are
% set in bold roman rather than the style's small caps, and the author
% grid runs three across rather than two.
%
% The style ships two running heads, "Under review as a conference paper
% at ICLR 2027" and "Published as a conference paper at ICLR 2027".
% Neither is true here, so \lhead is overridden to "Preprint".
%
% Style files are vendored in this folder: tectonic ignores TEXINPUTS,
% and arXiv provides only what TeX Live ships.
%
% Engine: pdflatex, which the ICLR style and the times package expect.
% No fontspec here, unlike the CSUR build. The cost is that fi and fl in
% T1-encoded Type1 fonts carry no Unicode mapping, so cmap is loaded to
% emit ToUnicode CMaps and keep the text searchable. cmap is a pdfTeX
% package: it works on arXiv and is inert under the local tectonic
% build, which is XeTeX.

% body.tex uses \rowcolors, which needs xcolor's table option in place
% before xcolor loads. Must precede \documentclass.
\PassOptionsToPackage{table}{xcolor}

% cmap must precede fontenc.
\usepackage{cmap}
\usepackage[T1]{fontenc}

\usepackage{microtype}

% The ICLR style sets the title, the abstract label and every heading
% level in small caps. Redefined below to bold roman, which reads as a
% paper rather than as a typeset proceeding. Spacing and sizes are the
% style's own values, carried over unchanged, so only the face differs.
\makeatletter
% The style sets the title, the abstract label and every heading in
% small caps. Redefined here to bold roman, so the paper reads as a
% paper rather than as a typeset proceeding. Sizes and spacing are the
% style's own values, carried over unchanged; only the face differs.
%
% \@maketitle is reproduced from the style's \ificlrfinal branch, which
% is the branch we always take, with \sc replaced by \bfseries and the
% 24pt strut reduced so three author columns fit on a row instead of
% two. The columns carry name and email only: all five share one
% affiliation, so it is stated once beneath the grid rather than five
% times inside it, which is also what makes three fit.
\def\@maketitle{\vbox{\hsize\textwidth
  {\LARGE\bfseries \@title\par}
  \vskip 0.22in
  \@author
  \vskip 0.12in}}
\renewenvironment{abstract}%
  {\vskip.075in\centerline{\large\bfseries Abstract}\vspace{0.5ex}\begin{quote}}%
  {\par\end{quote}\vskip 1ex}
\def\section{\@startsection{section}{1}{\z@}{-2.0ex plus -0.5ex minus -.2ex}%
  {1.5ex plus 0.3ex minus 0.2ex}{\large\bfseries\raggedright}}
\def\subsection{\@startsection{subsection}{2}{\z@}{-1.8ex plus -0.5ex minus -.2ex}%
  {0.8ex plus .2ex}{\normalsize\bfseries\raggedright}}
\def\subsubsection{\@startsection{subsubsection}{3}{\z@}{-1.5ex plus -0.5ex minus -.2ex}%
  {0.5ex plus .2ex}{\normalsize\bfseries\raggedright}}
\def\paragraph{\@startsection{paragraph}{4}{\z@}{1.5ex plus 0.5ex minus .2ex}%
  {-1em}{\normalsize\bfseries}}
\makeatother
